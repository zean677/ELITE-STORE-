version: "3.9"

services:
  # ─── PostgreSQL Database ─────────────────────────────────────────────────────
  postgres:
    image: postgres:16-alpine
    container_name: elite_store_db
    restart: unless-stopped
    environment:
      POSTGRES_USER: ${POSTGRES_USER:-elite}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB:-elite_store}
    volumes:
      - postgres_data:/var/lib/postgresql/data
    healthcheck:
      test: ["CMD-SHELL", "pg_isready -U ${POSTGRES_USER:-elite} -d ${POSTGRES_DB:-elite_store}"]
      interval: 10s
      timeout: 5s
      retries: 5
    networks:
      - elite_net

  # ─── Elite Store App (API + Frontend + Telegram Bot) ─────────────────────────
  app:
    build:
      context: .
      dockerfile: Dockerfile
    container_name: elite_store_app
    restart: unless-stopped
    depends_on:
      postgres:
        condition: service_healthy
    environment:
      NODE_ENV: production
      PORT: 8080
      DATABASE_URL: postgresql://${POSTGRES_USER:-elite}:${POSTGRES_PASSWORD}@postgres:5432/${POSTGRES_DB:-elite_store}
      SESSION_SECRET: ${SESSION_SECRET}
      TELEGRAM_BOT_TOKEN: ${TELEGRAM_BOT_TOKEN}
      TELEGRAM_ADMIN_CHAT_ID: ${TELEGRAM_ADMIN_CHAT_ID}
      TELEGRAM_WEBHOOK_URL: ${TELEGRAM_WEBHOOK_URL:-}
      SITE_URL: ${SITE_URL:-}
      ALLOWED_ORIGINS: ${ALLOWED_ORIGINS:-}
      SHAMS4STORE_API_TOKEN: ${SHAMS4STORE_API_TOKEN:-}
      AHMINIX_API_KEY: ${AHMINIX_API_KEY:-}
    ports:
      - "8080:8080"
    networks:
      - elite_net
    healthcheck:
      test: ["CMD-SHELL", "wget -qO- http://localhost:8080/health || exit 1"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 30s

volumes:
  postgres_data:
    driver: local

networks:
  elite_net:
    driver: bridge
